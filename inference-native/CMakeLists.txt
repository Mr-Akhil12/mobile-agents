cmake_minimum_required(VERSION 3.22)
project(lfm_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS lfm_compile=2 lfm_link=1)
set(CMAKE_JOB_POOL_COMPILE lfm_compile)
set(CMAKE_JOB_POOL_LINK lfm_link)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama-build)
add_library(lfm_core STATIC src/main/cpp/runtime.cpp src/main/cpp/omni.cpp)
target_link_libraries(lfm_core PUBLIC llama llama-common mtmd)
target_include_directories(lfm_core PUBLIC src/main/cpp)
if(ANDROID)
    add_library(lfm_jni SHARED src/main/cpp/bridge.cpp)
    target_link_libraries(lfm_jni PRIVATE lfm_core android log)
    target_link_options(lfm_jni PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(lfm-native-smoke src/test/cpp/smoke.cpp)
    target_link_libraries(lfm-native-smoke PRIVATE lfm_core)
endif()
